// E2E-only local TURN relay standing in for VK's (scripts/qwdtt-e2e.sh copies
// it into the qWDTT client's module as turnsrv/main.go). Never shipped.
package main

import (
	"log"
	"net"

	"github.com/pion/turn/v5"
)

func main() {
	pc, err := net.ListenPacket("udp4", "127.0.0.1:3478")
	if err != nil {
		log.Fatal(err)
	}
	key := turn.GenerateAuthKey("e2e", "e2e.local", "e2e")
	s, err := turn.NewServer(turn.ServerConfig{
		Realm: "e2e.local",
		AuthHandler: func(ra *turn.RequestAttributes) (string, []byte, bool) {
			return "e2e", key, ra.Username == "e2e"
		},
		PacketConnConfigs: []turn.PacketConnConfig{{
			PacketConn:            pc,
			RelayAddressGenerator: &turn.RelayAddressGeneratorStatic{RelayAddress: net.ParseIP("127.0.0.1"), Address: "127.0.0.1"},
		}},
	})
	if err != nil {
		log.Fatal(err)
	}
	log.Println("turn on 127.0.0.1:3478")
	defer s.Close()
	select {}
}
